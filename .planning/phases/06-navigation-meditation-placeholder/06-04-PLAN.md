      try {
        let duration: number;
        try {

        } catch {
          setError('Could not read file');
          return;

          filename: file.name,
          duration,
          blob: file,
          addedAt: Date.now(),

          // Prepend — newest-first matches plan 06-03 sort order.
          setMeditations(prev => [record, ...prev]);

            setError('Not enough storage');
            throw err;
          }
          setError('Could not save meditation');
          throw err;
        }
      } finally {
        setAdding(false);
      }
