```typescript
import { useRef, useCallback } from 'react';
import type { BreathingState, BreathingAction, BreathingConfig } from '../types/breathing';
import { DEFAULT_CONFIG, MIN_SECONDS, MAX_SECONDS } from '../types/breathing';

// --- Exported reducer (tested independently) ---
export function breathingReducer(state: BreathingState, action: BreathingAction): BreathingState {
  switch (action.type) {
    case 'START':
      if (state.session === 'running') return state;
      return { ...state, session: 'running' };
    case 'STOP':
      if (state.session === 'idle') return state;
      return { ...state, session: 'idle' };
    case 'SET_INHALE': {
      if (state.session === 'running') return state;
      const value = Math.max(MIN_SECONDS.inhale, Math.min(MAX_SECONDS.inhale, action.value));
      return { ...state, config: { ...state.config, inhale: value } };
    }
    case 'SET_HOLD': {
      if (state.session === 'running') return state;
      const value = Math.max(MIN_SECONDS.hold, Math.min(MAX_SECONDS.hold, action.value));
      return { ...state, config: { ...state.config, hold: value } };
    }
    case 'SET_EXHALE': {
      if (state.session === 'running') return state;
      const value = Math.max(MIN_SECONDS.exhale, Math.min(MAX_SECONDS.exhale, action.value));
      return { ...state, config: { ...state.config, exhale: value } };
    }
    default:
      return state;
  }
}

// --- Animation hook ---
export function useBreathingAnimation() {
  const animationRef = useRef<Animation | null>(null);
  const elementRef = useRef<HTMLDivElement | null>(null);
  const configRef = useRef<BreathingConfig>({ ...DEFAULT_CONFIG });
  const runningRef = useRef(false);

  const startCycle = useCallback(() => {
    const el = elementRef.current;
    if (!el || !runningRef.current) return;

    const { inhale, hold, exhale } = configRef.current;
    const totalMs = (inhale + hold + exhale) * 1000;
    const inhaleEnd = inhale / (inhale + hold + exhale);
    const holdEnd = inhaleEnd + hold / (inhale + hold + exhale);

    const maxScale = 220 / 48; // ~4.583 per D-01

    // Build keyframes — skip hold keyframe when hold=0 per D-07
    const keyframes: Keyframe[] = hold > 0
      ? [
          { transform: 'scale(1)', offset: 0, easing: 'ease-in-out' },
          { transform: `scale(${maxScale})`, offset: inhaleEnd, easing: 'linear' },
          { transform: `scale(${maxScale})`, offset: holdEnd, easing: 'ease-in-out' },
          { transform: 'scale(1)', offset: 1 },
        ]
      : [
          { transform: 'scale(1)', offset: 0, easing: 'ease-in-out' },
          { transform: `scale(${maxScale})`, offset: inhaleEnd, easing: 'ease-in-out' },
          { transform: 'scale(1)', offset: 1 },
        ];

    const anim = el.animate(keyframes, {
      duration: totalMs,
      iterations: 1,
      fill: 'forwards',
    });

    anim.onfinish = () => {
      // Loop: restart cycle (picks up any config changes via configRef)
      startCycle();
    };

    animationRef.current = anim;
  }, []);

  const start = useCallback((config: BreathingConfig) => {
    configRef.current = config;
    runningRef.current = true;
    startCycle();
  }, [startCycle]);

  const stop = useCallback(() => {
    runningRef.current = false;
    if (animationRef.current) {
      animationRef.current.cancel();
      animationRef.current = null;
    }
    // Reset scale to 1 (48px natural size)
    if (elementRef.current) {
      elementRef.current.style.transform = 'scale(1)';
    }
  }, []);

  return { elementRef, start, stop };
}
```

Key design decisions:
- `breathingReducer` is exported separately so it can be tested without rendering (per test expectations)
- `configRef` allows mid-session config updates to take effect on next cycle (per D-08 + Pattern 4)
- `runningRef` prevents orphaned onfinish callbacks from restarting after stop
- `fill: 'forwards'` prevents flash-back to scale(1) between cycles
- Element base size is 48px (set via CSS `w-12 h-12`), animation scales up to 220/48 = 4.583x per D-01
- When hold=0, hold keyframe is filtered out per D-07 to avoid interpolation artifacts (Pitfall 5)

```typescript
import { forwardRef } from 'react';
import type { SessionState } from '../types/breathing';

interface BreathingCircleProps {
  session: SessionState;
  onClick: () => void;
}

export const BreathingCircle = forwardRef<HTMLDivElement, BreathingCircleProps>(
  function BreathingCircle({ session, onClick }, ref) {
    const isIdle = session === 'idle';

    return (
      <div
        ref={ref}
        data-testid="breathing-circle"
        role="button"
        aria-label={isIdle ? 'Start breathing session' : 'Stop breathing session'}
        tabIndex={0}
        onClick={onClick}
        onKeyDown={(e) => { if (e.key === 'Enter' || e.key === ' ') onClick(); }}
        className={`w-12 h-12 rounded-full bg-white cursor-pointer transition-shadow duration-300 ${
          isIdle
            ? 'shadow-[0_0_20px_rgba(255,255,255,0.4)]'
            : 'shadow-none'
        }`}
        style={{ transformOrigin: 'center', willChange: 'transform' }}
      />
    );
  }
);
```

Key points per decisions:
- `w-12 h-12` = 48px base size per D-01
- `shadow-[0_0_20px_rgba(255,255,255,0.4)]` = subtle glow in idle per D-11
- `shadow-none` during session per D-12
- `transformOrigin: 'center'` prevents asymmetric scaling (Pitfall 3)
- `will-change: transform` promotes to GPU compositor layer
- `forwardRef` so parent can pass elementRef from the animation hook
- `role="button"` + `tabIndex` + `onKeyDown` for accessibility

```typescript
import { useReducer } from 'react';
import { BreathingCircle } from './BreathingCircle';
import { useBreathingAnimation, breathingReducer } from '../hooks/useBreathingAnimation';
import { DEFAULT_CONFIG } from '../types/breathing';
import type { BreathingState } from '../types/breathing';

const initialState: BreathingState = {
  session: 'idle',
  config: { ...DEFAULT_CONFIG },
};

export function BreathingScreen() {
  const [state, dispatch] = useReducer(breathingReducer, initialState);
  const { elementRef, start, stop } = useBreathingAnimation();

  const handleCircleTap = () => {
    if (state.session === 'idle') {
      dispatch({ type: 'START' });
      start(state.config);
    } else {
      dispatch({ type: 'STOP' });
      stop();
    }
  };

  return (
    <div
      data-panel="breathing"
      data-session={state.session}
      aria-label="Breathing"
      className="w-screen h-dvh flex-shrink-0 snap-start snap-always flex items-center justify-center bg-black relative"
    >
      <BreathingCircle
        ref={elementRef}
        session={state.session}
        onClick={handleCircleTap}
      />
    </div>
  );
}
```

Key points:
- `data-panel="breathing"` preserved (navigation test contract)
- `data-session={state.session}` added (breathing-screen test expects this)
- `relative` added to outer div for future absolute-positioned controls (Plan 03)
- Circle centered via `flex items-center justify-center` per D-14
- No controls yet — they come in Plan 03. This plan delivers the core tap-to-animate experience.
