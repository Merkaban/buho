```typescript
export type SessionState = 'idle' | 'running';

export interface BreathingConfig {
  inhale: number; // seconds (min: 1, max: 60)
  hold: number;   // seconds (min: 0, max: 60)
  exhale: number; // seconds (min: 1, max: 60)
}

export interface BreathingState {
  session: SessionState;
  config: BreathingConfig;
}

export type BreathingAction =
  | { type: 'START' }
  | { type: 'STOP' }
  | { type: 'SET_INHALE'; value: number }
  | { type: 'SET_HOLD'; value: number }
  | { type: 'SET_EXHALE'; value: number };

export const DEFAULT_CONFIG: BreathingConfig = { inhale: 5, hold: 5, exhale: 5 };
export const MIN_SECONDS = { inhale: 1, hold: 0, exhale: 1 };
export const MAX_SECONDS = { inhale: 60, hold: 60, exhale: 60 };
```

From src/components/BreathingScreen.tsx (after Plan 02):
```typescript
import { useReducer } from 'react';
import { BreathingCircle } from './BreathingCircle';
import { useBreathingAnimation, breathingReducer } from '../hooks/useBreathingAnimation';
import { DEFAULT_CONFIG } from '../types/breathing';
import type { BreathingState } from '../types/breathing';

const initialState: BreathingState = {
  session: 'idle',
  config: { ...DEFAULT_CONFIG },
};

export function BreathingScreen() {
  const [state, dispatch] = useReducer(breathingReducer, initialState);
  const { elementRef, start, stop } = useBreathingAnimation();

  const handleCircleTap = () => {
    if (state.session === 'idle') {
      dispatch({ type: 'START' });
      start(state.config);
    } else {
      dispatch({ type: 'STOP' });
      stop();
    }
  };

  return (
    <div
      data-panel="breathing"
      data-session={state.session}
      aria-label="Breathing"
      className="w-screen h-dvh flex-shrink-0 snap-start snap-always flex items-center justify-center bg-black relative"
    >
      <BreathingCircle
        ref={elementRef}
        session={state.session}
        onClick={handleCircleTap}
      />
    </div>
  );
}
```

The controls dispatch SET_INHALE, SET_HOLD, SET_EXHALE actions. They receive `config` to display current values and a `dispatch` function. They also receive `visible` boolean for the opacity fade.

```typescript
import type { BreathingConfig, BreathingAction } from '../types/breathing';
import { MIN_SECONDS, MAX_SECONDS } from '../types/breathing';

interface StepperProps {
  label: string;
  value: number;
  min: number;
  max: number;
  onChange: (value: number) => void;
}

function Stepper({ label, value, min, max, onChange }: StepperProps) {
  return (
    <div className="flex flex-col items-center gap-1">
      <span className="text-white/60 text-xs uppercase tracking-wider">{label}</span>
      <div className="flex items-center gap-3">
        <button
          onClick={() => onChange(value - 1)}
          disabled={value <= min}
          className="w-8 h-8 rounded-full border border-white/30 text-white flex items-center justify-center disabled:opacity-30 active:bg-white/10"
          aria-label={`Decrease ${label}`}
        >
          −
        </button>
        <span className="text-white text-lg w-6 text-center tabular-nums">{value}</span>
        <button
          onClick={() => onChange(value + 1)}
          disabled={value >= max}
          className="w-8 h-8 rounded-full border border-white/30 text-white flex items-center justify-center disabled:opacity-30 active:bg-white/10"
          aria-label={`Increase ${label}`}
        >
          +
        </button>
      </div>
    </div>
  );
}

interface TimingControlsProps {
  config: BreathingConfig;
  dispatch: React.Dispatch<BreathingAction>;
  visible: boolean;
}

export function TimingControls({ config, dispatch, visible }: TimingControlsProps) {
  return (
    <div
      data-testid="timing-controls"
      className={`absolute top-[15%] left-0 right-0 flex justify-center gap-8 transition-opacity duration-500 ${
        visible ? 'opacity-100' : 'opacity-0 pointer-events-none'
      }`}
      aria-hidden={!visible}
    >
      <Stepper
        label="inhale"
        value={config.inhale}
        min={MIN_SECONDS.inhale}
        max={MAX_SECONDS.inhale}
        onChange={(v) => dispatch({ type: 'SET_INHALE', value: v })}
      />
      <Stepper
        label="hold"
        value={config.hold}
        min={MIN_SECONDS.hold}
        max={MAX_SECONDS.hold}
        onChange={(v) => dispatch({ type: 'SET_HOLD', value: v })}
      />
      <Stepper
        label="exhale"
        value={config.exhale}
        min={MIN_SECONDS.exhale}
        max={MAX_SECONDS.exhale}
        onChange={(v) => dispatch({ type: 'SET_EXHALE', value: v })}
      />
    </div>
  );
}
```

Design decisions per CONTEXT.md:
- `absolute top-[15%]` positions controls in upper third per D-13 without affecting circle centering (D-14, D-17)
- `transition-opacity duration-500` for smooth fade per D-15
- `opacity-0 pointer-events-none` when hidden — element stays in DOM, no layout shift per D-17
- `aria-hidden={!visible}` hides from screen readers when faded
- `tabular-nums` prevents width jumping as numbers change
- `active:bg-white/10` gives touch feedback
- Minus button uses `−` (proper minus sign U+2212) not hyphen
- Stepper calls `onChange(value - 1)` / `onChange(value + 1)` — the reducer handles clamping, so the component doesn't need to
