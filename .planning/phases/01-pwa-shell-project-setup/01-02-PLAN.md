1. Create `src/components/SwipeContainer.tsx`:
```typescript
import { useRef, useCallback, type ReactNode } from 'react';

interface SwipeContainerProps {
  children: ReactNode;
  onPageChange: (index: number) => void;
}

export function SwipeContainer({ children, onPageChange }: SwipeContainerProps) {
  const containerRef = useRef<HTMLDivElement>(null);
  const lastIndexRef = useRef(0);

  const handleScroll = useCallback(() => {
    const el = containerRef.current;
    if (!el) return;
    const index = Math.round(el.scrollLeft / el.clientWidth);
    if (index !== lastIndexRef.current) {
      lastIndexRef.current = index;
      onPageChange(index);
    }
  }, [onPageChange]);

  return (
    <div
      ref={containerRef}
      onScroll={handleScroll}
      className="scroll-container flex w-screen h-dvh overflow-x-auto snap-x snap-mandatory"
    >
      {children}
    </div>
  );
}
```

Key details:
- Uses the `.scroll-container` class from App.css which has `overscroll-behavior-x: none`, `-webkit-overflow-scrolling: touch`, and hidden scrollbar styles
- `lastIndexRef` prevents redundant `onPageChange` calls during scroll (Pitfall 4 from RESEARCH.md)
- `snap-x snap-mandatory` provides CSS scroll-snap behavior
- `h-dvh` uses dynamic viewport height (not `h-screen` which is `100vh` and broken on iOS)

  );
}

```typescript
import { useState } from 'react';
import './App.css';
import { SwipeContainer } from './components/SwipeContainer';
import { PageDots } from './components/PageDots';
import { LandingScreen } from './components/LandingScreen';
import { BreathingScreen } from './components/BreathingScreen';
import { MeditationScreen } from './components/MeditationScreen';

function App() {
  const [activePage, setActivePage] = useState(0);

  return (
    <div
      data-safe-area
      className="w-screen h-dvh bg-black"
      style={{
        paddingTop: 'env(safe-area-inset-top)',
        paddingLeft: 'env(safe-area-inset-left)',
        paddingRight: 'env(safe-area-inset-right)',
      }}
    >
      <SwipeContainer onPageChange={setActivePage}>
        <LandingScreen />
        <BreathingScreen />
        <MeditationScreen />
      </SwipeContainer>
      <PageDots total={3} active={activePage} />
    </div>
  );
}

export default App;
```

Key details:
- `data-safe-area` attribute matches test expectation in navigation.test.tsx
- Safe area insets applied via inline style (top, left, right) per UI-SPEC Layout Contract. Bottom handled by PageDots component itself.
- `useState(0)` tracks active panel index, passed to PageDots
- `setActivePage` passed directly as `onPageChange` callback to SwipeContainer
- All 3 screen components rendered as children of SwipeContainer
- PageDots positioned fixed (in its own component), outside the scroll container
- bg-black on wrapper prevents any non-black flash between panels
- h-dvh ensures full dynamic viewport height (iOS compatible)
